cmake_minimum_required(VERSION 3.20)
project(gluspot_airplay LANGUAGES C CXX)
if(NOT WIN32)
    message(FATAL_ERROR "GluSpotAirPlay requires Windows WASAPI")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(AIRPLAY_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
set(AIRPLAY_BUILD_QT_HOST OFF CACHE BOOL "" FORCE)
option(GLUSPOT_AIRPLAY_TESTS "Build native protocol and PCM tests" ON)
set(AIRPLAY_BUILD_TESTS ${GLUSPOT_AIRPLAY_TESTS} CACHE BOOL "" FORCE)
if(GLUSPOT_AIRPLAY_TESTS)
    enable_testing()
endif()
add_subdirectory(../../third_party/airplay2-sender-cpp upstream)
add_executable(GluSpotAirPlay main.cpp)
target_link_libraries(GluSpotAirPlay PRIVATE raop_sender ole32 uuid avrt ksuser)
target_compile_definitions(GluSpotAirPlay PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
    target_compile_options(GluSpotAirPlay PRIVATE /W4 /utf-8)
else()
    target_compile_options(GluSpotAirPlay PRIVATE -Wall -Wextra)
    target_link_options(GluSpotAirPlay PRIVATE -static -municode)
endif()
set_target_properties(GluSpotAirPlay PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(GluSpotAirPlay PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}")
endforeach()
if(GLUSPOT_AIRPLAY_TESTS)
    add_executable(gluspot_pcm_tests pcm_tests.cpp)
    add_executable(gluspot_timing_tests timing_tests.cpp)
    add_executable(gluspot_command_tests command_tests.cpp)
    add_executable(gluspot_sender_policy_tests sender_policy_tests.cpp)
    target_link_libraries(gluspot_sender_policy_tests PRIVATE raop_sender)
    if(MSVC)
        target_compile_options(gluspot_pcm_tests PRIVATE /UNDEBUG)
        target_compile_options(gluspot_timing_tests PRIVATE /UNDEBUG)
        target_compile_options(gluspot_command_tests PRIVATE /UNDEBUG)
    else()
        target_compile_options(gluspot_pcm_tests PRIVATE -UNDEBUG)
        target_compile_options(gluspot_timing_tests PRIVATE -UNDEBUG)
        target_compile_options(gluspot_command_tests PRIVATE -UNDEBUG)
        target_link_options(gluspot_pcm_tests PRIVATE -static)
        target_link_options(gluspot_timing_tests PRIVATE -static)
        target_link_options(gluspot_command_tests PRIVATE -static)
    endif()
    # The upstream loopback tests also need to run without a compiler DLL path.
    if(NOT MSVC)
        target_link_options(gluspot_sender_policy_tests PRIVATE -static)
        target_link_options(raop_core_tests PRIVATE -static)
        target_link_options(raop_loop_tests PRIVATE -static)
    endif()
    add_test(NAME gluspot_pcm_tests COMMAND gluspot_pcm_tests)
    add_test(NAME gluspot_timing_tests COMMAND gluspot_timing_tests)
    add_test(NAME gluspot_command_tests COMMAND gluspot_command_tests)
    add_test(NAME gluspot_sender_policy_tests COMMAND gluspot_sender_policy_tests)
endif()
